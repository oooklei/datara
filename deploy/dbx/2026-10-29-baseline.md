```yaml
    container_name: dbx

      - "4224:4224"
    environment:
      - DBX_PASSWORD=${DBX_PASSWORD}
      - DBX_BACKUP_ROOT=/app/backups
      - TZ=Asia/Shanghai
    volumes:
      - /lei/apps/dbx/data:/app/data
      - /lei/apps/dbx/backups:/app/backups
    restart: unless-stopped
    stop_grace_period: 90s
```
