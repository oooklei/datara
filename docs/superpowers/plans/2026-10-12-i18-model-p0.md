```python
"""I18 DDL 方言生成器（纯函数）：mysql / greatdb / doris。

- 标识符一律经 dsconn.quote_ident 校验（与 meta_ops 同注入标准）
- 长度渲染：VARCHAR(n)/CHAR(n)/DECIMAL(p,s) 带参，其余裸类型大写
- is_partition 仅 doris 生效；mysql/greatdb 由 dialect_notes 提示不支持
- greatdb 与 mysql 同语系（dsconn 裁定②：GreatDB 走 MySQL 协议），无额外参数
- 字段入参形状为 API 契约 camelCase（见 api/model.py _field_row）
"""

from typing import Optional

from common.dsconn import quote_ident


def _lit(v: Optional[str]) -> str:
    """字符串字面量（单引号转义）。"""
    return "'" + str(v).replace("'", "''") + "'"


def _ty(f: dict) -> str:
    t = str(f.get("dataType") or "VARCHAR").upper()
    length = str(f.get("length") or "")
    precision = str(f.get("precision") or "")
    if t in ("VARCHAR", "CHAR") and length:
        return f"{t}({length})"
    if t == "DECIMAL" and length:
        return f"DECIMAL({length},{precision})" if precision else f"DECIMAL({length})"
    return t


def dialect_notes(fields: list[dict], engine: str) -> list[str]:
    """方言提示（前端预览区展示）。"""
    notes: list[str] = []
    if engine != "doris" and any(f.get("isPartition") for f in fields):
        notes.append("该引擎不支持分区声明")
    return notes


def gen_ddl(code: str, name: str, fields: list[dict], engine: str) -> str:
    """按方言生成 CREATE TABLE 文本（不落库）。

    标识符不合法直接 ValueError（路由转 400 MODEL_CODE_INVALID）。
    """
    col_lines: list[str] = []
    pk_parts: list[str] = []
    part_parts: list[str] = []
    for f in fields:
        ident = quote_ident(str(f.get("name") or ""))
        seg = f"{ident} {_ty(f)}"
        if not f.get("nullable", True):
            seg += " NOT NULL"
        if f.get("defaultValue"):
            seg += f" DEFAULT {_lit(f['defaultValue'])}"
        seg += f" COMMENT {_lit(f.get('comment') or f.get('name') or '')}"
        col_lines.append(seg)
        if f.get("isPk"):
            pk_parts.append(ident)
        if f.get("isPartition"):
            part_parts.append(ident)

    lines = col_lines + (["PRIMARY KEY (" + ", ".join(pk_parts) + ")"] if pk_parts else [])
    body = ",\n  ".join(lines)

    if engine == "doris":
        head = (
            f"CREATE TABLE {quote_ident(code)} (\n  {body}\n)\n"
        )
        if part_parts:
            head += f"PARTITION BY RANGE({', '.join(part_parts)}) ()\n"
        dist = ", ".join(pk_parts) if pk_parts else quote_ident(str(fields[0]["name"]))
        return (
            head
            + f"DISTRIBUTED BY HASH({dist}) BUCKETS 10\n"
            + "PROPERTIES (\n"
            + "  'replication_num' = '1',\n"
            + "  'dynamic_partition.enable' = 'true',\n"
            + "  'dynamic_partition.time_unit' = 'DAY',\n"
            + "  'dynamic_partition.start' = '-30',\n"
            + "  'dynamic_partition.end' = '3',\n"
            + "  'dynamic_partition.prefix' = 'p'\n"
            + ")"
        )

    # mysql / greatdb（同语系）
    return (
        f"CREATE TABLE {quote_ident(code)} (\n  {body}\n"
        f") ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT={_lit(name)}"
    )
```

```python
def parse_create_tables(sql: str) -> list[dict]:
    """I18 逆向导入（DDL 文件）：解析 CREATE TABLE → [{name, comment, fields[]}]。

    - sqlglot read='mysql'；语法错误上抛 sqlglot.errors.ParseError（路由转 400 附行号）
    - 字段形状为 API 契约 camelCase；仅取 CREATE TABLE，其余语句忽略
    """
    from sqlglot import exp

    out: list[dict] = []
    for stmt in sqlglot.parse(sql, read="mysql") or []:
        if not (isinstance(stmt, exp.Create) and stmt.kind == "TABLE"):
            continue
        schema = stmt.this
        if not isinstance(schema, exp.Schema):
            continue
        name = schema.this.name
        comment = ""
        if stmt.props:
            for p in stmt.props.properties:
                pname = getattr(p, "name", "") or ""
                if pname.upper() == "COMMENT":
                    comment = str(p.args.get("value") or "")
        pk_cols: set[str] = set()
        for node in schema.expressions:
            if isinstance(node, exp.PrimaryKey):
                pk_cols.update(c.name for c in node.expressions)
        fields = []
        for coldef in schema.find_all(exp.ColumnDef):
            kind = coldef.kind
            dtype = (kind.name if kind is not None else "VARCHAR").upper()
            params = [e.sql() for e in (kind.expressions if kind is not None else [])]
            nullable = not any(
                getattr(c.this, "key", "") == "notnullconstraint"
                for c in (coldef.args.get("constraints") or [])
            )
            d = coldef.args.get("default")
            fields.append({
                "name": coldef.name,
                "dataType": dtype,
                "length": str(params[0]) if params else "",
                "precision": str(params[1]) if len(params) > 1 else "",
                "nullable": nullable,
                "defaultValue": str(d.sql().strip("'")) if d is not None else "",
                "comment": " ".join(coldef.args.get("comments") or []),
                "isPk": coldef.name in pk_cols,
                "isPartition": False,
                "dictRef": "",
            })
        out.append({"name": name, "comment": comment, "fields": fields})
    return out
```

注意：`ParseError` 带 `line`/`col` 属性；sqlglot 版本间 ColumnDef 约束/注释的 args 键名可能略有差异，**执行以失败用例的实际断言输出为准微调取值键**（TDD 红灯信息即依据），不得改测试期望语义。

```python
@router.get("/models")
def list_models(
    layer: str = "", status: str = "", domain: str = "", keyword: str = "",
    page: int = 1, size: int = 50,
    user: User = Depends(get_current_user), db: Session = Depends(get_db),
):

    kw = keyword.strip()

    def _base(q):
        if status:
            q = q.filter(DwModel.status == status)
        if kw:
            like = f"%{kw}%"
            q = q.filter(or_(DwModel.code.like(like), DwModel.name.like(like), DwModel.owner.like(like)))
        return q

    q = _base(db.query(DwModel))
    if layer:
        q = q.filter(DwModel.layer == layer)
    if domain:
        q = q.filter(DwModel.biz_domain == domain)
    total = q.count()
    rows = q.order_by(DwModel.update_time.desc()).offset((page - 1) * size).limit(size).all()
    xcs = _xc_map(db)

    layer_stats = []
    for l in db.query(DwLayer).order_by(DwLayer.seq).all():

        if layer:
            c = c.filter(DwModel.layer == layer)
        domain_stats.append({"name": d.name, "count": c.scalar()})

               "layers": layer_stats, "domains": domain_stats})
```

```python
class PhysicalizeBody(BaseModel):
    engine: str
    dsId: Optional[int] = None
    dbName: Optional[str] = None
    mode: str = "script"  # exec | script
    ddl: Optional[str] = None


@router.post("/models/{model_id}/physicalize")
def physicalize(model_id: int, body: PhysicalizeBody,
                user: User = Depends(require_perm("model:write")), db: Session = Depends(get_db)):
    """物理化（R3 复用 dsconn 连接；60s 读超时对齐 meta_ops）。

    - script：仅返回校验后脚本文本，状态不变
    - exec：真实连接执行（sqlglot 拆语句防注释内分号误拆），成功→built，
      失败→状态不变并附已执行条数（I18 §8）
    """
    m = _get_model(db, model_id)
    fields = _fields_of(db, model_id)
    sql = body.ddl or gen_ddl(m.code, m.name, fields, body.engine)
    if body.mode == "script":
        return ok({"script": sql, "notes": dialect_notes(fields, body.engine)})

    ds = db.get(DataSource, body.dsId) if body.dsId else None
    if ds is None or ds.type not in ("mysql", "greatdb"):
        raise ApiError(DS_NOT_FOUND, "物理化需选择 mysql/greatdb 连接型数据源", status=400)
    stmts = [s for s in (sqlglot.parse(sql, read="mysql") or []) if s is not None]
    executed = 0
    try:
        conn = open_connection(ds, db=body.dbName or None)
        try:
            with conn.cursor() as cur:
                for s in stmts:
                    cur.execute(s.sql(dialect="mysql"))
                    executed += 1
                conn.commit()
        finally:
            conn.close()
    except Exception as exc:  # noqa: BLE001 执行失败：状态不变 + 审计 + 已执行条数
        audit.error("model.physicalize user=%s code=%s FAIL executed=%d err=%s",
                    user.user_name, m.code, executed, exc)
        raise ApiError(MODEL_PHYS_FAIL, f"DDL 执行失败（已执行 {executed} 条）：{exc}",
                       status=400, data={"executed": executed, "error": str(exc)})
    m.status = "built"
    m.last_ddl = sql
    m.built_at = now()
    m.built_by = user.user_name
    db.commit()
    audit.info("model.physicalize user=%s code=%s version=%s", user.user_name, m.code, m.version)
    return ok({"status": "built", "executed": executed})
```

```python
    if ds.type not in ("mysql", "greatdb"):
        raise ApiError(PARAM_INVALID, "逆向导入仅支持连接型数据源（mysql/greatdb）", status=400)

def _col_to_field(c: tuple) -> dict:
    """information_schema.COLUMNS 行（10 列）→ API 契约字段 dict。"""
    (name, data_type, _column_type, is_nullable, default, cmt, key, char_len, num_p, num_s) = c
    dtype = str(data_type or "varchar").upper()
    length, precision = "", ""
    if dtype in ("VARCHAR", "CHAR"):
        length = str(char_len or "")
    elif dtype in ("DECIMAL", "NUMERIC"):
        length = str(num_p or "")
        precision = str(num_s or "")
    return {
        "name": name, "dataType": dtype, "length": length, "precision": precision,
        "nullable": is_nullable == "YES",
        "defaultValue": str(default) if default is not None else "",
        "comment": cmt or "", "isPk": key == "PRI", "isPartition": False, "dictRef": "",
    }
```

```python
def model_lineage(model_id: int, user: User = Depends(get_current_user), db: Session = Depends(get_db)):
    """合并图（R4）：declared=t_dw_model_relation 双向展开；runtime=t_lineage_edge 按 code 匹配。"""
    m = _get_model(db, model_id)
    nodes = [{"id": m.code, "kind": "model", "label": f"{m.code} {m.name}"}]
    edges: list[dict] = []

    rels = (db.query(DwModelRelation)
            .filter(or_(DwModelRelation.src_model_id == model_id, DwModelRelation.tgt_model_id == model_id))
            .all())
    rel_ids = {r.src_model_id for r in rels} | {r.tgt_model_id for r in rels}
    name_by_id = {x.id: x for x in db.query(DwModel).filter(DwModel.id.in_(rel_ids)).all()}
    for r in rels:
        s = name_by_id.get(r.src_model_id)
        t = name_by_id.get(r.tgt_model_id)
        if not s or not t:
            continue
        if s.id != model_id:
            nodes.append({"id": s.code, "kind": "model", "label": f"{s.code} {s.name}"})
        if t.id != model_id:
            nodes.append({"id": t.code, "kind": "model", "label": f"{t.code} {t.name}"})
        edges.append({"from": s.code, "to": t.code, "kind": "declared", "relType": r.rel_type or ""})

    like = f"%.{m.code}"
    run = (db.query(LineageEdge)
           .filter(or_(LineageEdge.from_table.like(like), LineageEdge.to_table.like(like)))
           .limit(200).all())
    for e in run:
        if e.from_table and not e.from_table.endswith(f".{m.code}"):
            nodes.append({"id": e.from_table, "kind": "ext", "label": e.from_table})
        if e.to_table and not e.to_table.endswith(f".{m.code}"):
            nodes.append({"id": e.to_table, "kind": "ext", "label": e.to_table})
        edges.append({"from": e.from_table, "to": e.to_table, "kind": "runtime"})

    seen: set[str] = set()
    uniq_nodes = []
    for n in nodes:
        if n["id"] in seen:
            continue
        seen.add(n["id"])
        uniq_nodes.append(n)
    return ok({
        "nodes": uniq_nodes,
        "edges": edges,
        "sources": {"declared": sum(1 for e in edges if e["kind"] == "declared"),

    """下游递归清单（声明血缘 BFS，环防护）。"""
    m = _get_model(db, model_id)
    nxt: dict[int, list[int]] = {}
    for r in db.query(DwModelRelation).all():
        nxt.setdefault(r.src_model_id, []).append(r.tgt_model_id)
    out: list[dict] = []
    path = [m]
    seen = {m.id}

    def dfs(cur: DwModel) -> None:
        for t in nxt.get(cur.id, []):
            if t in seen:
                continue
            seen.add(t)
            tm = db.get(DwModel, t)
            if tm is None:
                continue
            path.append(tm)
            out.append({"modelId": tm.id, "code": tm.code, "name": tm.name,
                        "path": " → ".join(x.code for x in path)})
            dfs(tm)
            path.pop()

    dfs(m)
    return ok({"list": out})
```
