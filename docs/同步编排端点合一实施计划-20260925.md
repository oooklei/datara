```ts
/** 拖边即引用（设计 §3.3）：连边 → target 的 inputs 追加 sourceId:port；删边 → 移除。
 * port = sourceHandle ?? ''；inputs 字段不存在则不处理（该节点无 upstream-refs 输入）。 */
export function onEdgeCreated(doc: { nodes: GNodeLike[]; edges: EdgeLike[] }, edge: EdgeLike): void {
  const tgt = doc.nodes.find((n) => n.id === edge.target)
  if (!tgt || !Array.isArray(tgt.data?.inputs)) return
  const ref = `${edge.source}:${edge.sourceHandle ?? ''}`
  if (!tgt.data.inputs.includes(ref)) tgt.data.inputs.push(ref)
}

export function onEdgeRemoved(doc: { nodes: GNodeLike[]; edges: EdgeLike[] }, edge: EdgeLike): void {
  const tgt = doc.nodes.find((n) => n.id === edge.target)
  if (!tgt || !Array.isArray(tgt.data?.inputs)) return
  const ref = `${edge.source}:${edge.sourceHandle ?? ''}`
  tgt.data.inputs = tgt.data.inputs.filter((r: string) => r !== ref)
}
```

（GNodeLike/EdgeLike 用文件内既有类型或最小结构类型 `{ id: string; data: Record<string, unknown> }`。）

```ts
/** 目标表基准探测（设计 §3.1）：库表树 { [schema]: string[] } → 按 matchType/matchPrefix（或目标表名）过滤 → 候选 [{schema, table}] */
export function probeMatchTables(
  tree: Record<string, string[]>, tgtTable: string, matchType: string, matchPrefix: string,
): { schema: string; table: string }[] {
  const pattern = matchType === 'prefix' ? (matchPrefix || tgtTable) : tgtTable
  const out: { schema: string; table: string }[] = []
  for (const schema of Object.keys(tree).sort()) {
    for (const t of tree[schema] ?? []) {
      if (matchType === 'prefix' ? t.startsWith(pattern) : t === pattern) out.push({ schema, table: t })
    }
  }
  return out
}
```

```python
def _materialize_sync_exec(self, doc: dict) -> dict:
    """设计态 → 运行态（设计 §3.2）：链上存在 endpoint_select 时，在 assert 入边处物化执行节点。
    assert 的全部入边 source 改连 sys_exec，sys_exec → assert；执行类型按 baseMode 分拣。"""
    ep_nodes = [n for n in doc.get("nodes", []) if n.get("type") == "endpoint_select"]
    if not ep_nodes:
        return doc
    base_mode = str((ep_nodes[0].get("data") or {}).get("baseMode") or "src_base")
    exec_type = "file_sync" if base_mode == "file_sync" else "sync"
    edges = doc.get("edges", [])
    asserts = [n for n in doc.get("nodes", []) if n.get("type") == "assert"]
    if not asserts:
        return doc
    nid = "sys_exec_" + str(uuid.uuid4())[:8]
    doc["nodes"].append({
        "id": nid, "type": exec_type,
        "data": {"name": "文件入仓" if exec_type == "file_sync" else "同步执行", "type": exec_type},
    })
    for a in asserts:
        for e in [e for e in edges if e.get("target") == a["id"]]:
            e["target"] = nid
        edges.append({"id": f"e_{nid}_{a['id']}", "source": nid, "target": a["id"],
                      "sourceHandle": "", "label": ""})
    return doc
```
