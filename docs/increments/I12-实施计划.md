```python
"""C26 通知执行器（I12）：webhook POST / 仅日志；触发时机由 master 判定后派发。"""
from worker.executor import ExecResult, register
from common.vars_render import render_text   # 变量渲染（I10 下沉件）
import json, time, urllib.request

@register("notify")
def run_notify(ctx, param: dict) -> ExecResult:
    url = (render_text(param.get("url", ""), ctx.vars) or "").strip()
    tpl = param.get("template", "") or ""
    msg = render_text(tpl, ctx.vars)
    ctx.log(f"[notify] 通道={param.get('channel','log')} url={'有' if url else '无'} 消息={msg}")
    if param.get("channel") == "webhook" and url:
        req = urllib.request.Request(url, data=json.dumps({"text": msg}).encode(),
                                     headers={"Content-Type": "application/json"})
        try:
            with urllib.request.urlopen(req, timeout=10) as r:
                ctx.log(f"[notify] webhook 响应 {r.status}")
        except Exception as e:                       # 通知失败默认不阻断（设计 §3.3）
            ctx.log(f"[notify] webhook 失败（不阻断）：{e}")
            if param.get("fail_hard"):
                return ExecResult(success=False, outputs={"error": str(e)}, status="failure")
    return ExecResult(success=True, outputs={"notified": True}, status="success")
```
