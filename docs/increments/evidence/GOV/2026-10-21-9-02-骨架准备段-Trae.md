## ① ScheduleRow 类型面草案（对齐 `_payload` 17 键）

GREEN 段实装时落 `scheduleApi.ts`（或 O2 裁定后的 `scheduleCrudApi.ts`）。17 键与后端约束逐键对照：

```ts
/** 定时计划行（对齐 datara-backend _payload 17 键驼峰面，fmt_dt 可 null 处标 | null） */
export type ScheduleState = 'offline' | 'online'

export type ScheduleMisfirePolicy = 'FIRE_ONCE' | 'SKIP' | 'FIRE_ALL_LIMITED'
export type ScheduleConcurrencyPolicy = 'ALLOW' | 'FORBID' | 'QUEUE'

export interface ScheduleRow {
  id: number
  wfCode: number
  name: string
  crontab: string
  startTime: string | null
  endTime: string | null
  state: ScheduleState
  /** 后端钳制 1-5（create L164：max(1, min(x, 5))） */
  priority: number
  workerGroup: string | null
  /** 后端 ≥0（L166） */
  failRetryTimes: number
  /** 后端 ≥1（L167） */
  failRetryInterval: number
  /** validate_policies L111 枚举 */
  misfirePolicy: ScheduleMisfirePolicy
  /** validate_policies L112 枚举 */
  concurrencyPolicy: ScheduleConcurrencyPolicy
  /** 后端 ≥1（L115-116 PARAM_INVALID） */
  maxRunningInstances: number
  /** ZoneInfo 可实例化（L117-120），表单默认 Asia/Shanghai */
  timezone: string
  createTime: string | null
  updateTime: string | null
}
```

```ts
/** scheduleCrudApi 契约方法面（7 方法 ↔ 后端 7 端点，前缀 /api/v1，单 BASE 不变） */
export interface ScheduleCrudContract {
  /** GET  /api/v1/workflow-definitions/{wf}/schedules（{wf} 兼容定义 id 与数字 code） */
  listSchedules: (wf: number | string) => Promise<ScheduleRow[]>
  /** POST /api/v1/workflow-definitions/{wf}/schedules（初始 offline，返回含预览） */
  createSchedule: (wf: number | string, input: ScheduleInput) => Promise<ScheduleRow>
  /** PUT  /api/v1/schedules/{schedule_id}（crontab 变更重校验；update_time 驱动 master 缓存重算） */
  updateSchedule: (scheduleId: number, input: ScheduleInput) => Promise<ScheduleRow>
  /** DELETE /api/v1/schedules/{schedule_id}（后端语义：先 offline 再删） */
  deleteSchedule: (scheduleId: number) => Promise<void>
  /** POST /api/v1/schedules/{schedule_id}/online（上线前 crontab 再校验） */
  onlineSchedule: (scheduleId: number) => Promise<ScheduleRow>
  /** POST /api/v1/schedules/{schedule_id}/offline */
  offlineSchedule: (scheduleId: number) => Promise<ScheduleRow>
  /** POST /api/v1/crontab-preview（timezone 感知；预览未通过禁止提交为 UI 硬约束） */
  previewCrontab: (crontab: string, timezone?: string) => Promise<CrontabPreview>
}
```
