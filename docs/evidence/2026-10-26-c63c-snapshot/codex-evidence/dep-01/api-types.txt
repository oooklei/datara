17:/* ---------- 周期 / 日期范围（DolphinScheduler 依赖窗口） ---------- */
18:
19:export type DependencyPeriodType = 'hour' | 'day' | 'week' | 'month'
20:
21:/** 相对上游任务计划时间的窗口：offset 0=当前周期，-1=上一周期（day/hour/week/month 各自解释） */
22:export interface DependencyDateRange {
23:  periodType: DependencyPeriodType
24:  beginOffset: number
25:  endOffset: number
26:}

28:/* ---------- 依赖条件与组 ---------- */
29:
30:export interface DependencyCondition {
31:  kind?: 'workflow' | 'task' | 'file' | 'table_state' | 'external_task'
32:  selector?: 'workflow' | 'all_tasks' | 'task'
33:  upstreamWorkflowId: number
34:  /** 固定上游版本；缺省表示随上游最新已发布版本（后端冻结前为草案语义） */
35:  upstreamWorkflowVersion?: number
36:  /** 可选：依赖上游某具体任务；缺省为整工作流成功 */
37:  upstreamTaskCode?: string
38:  dateRange?: DependencyDateRange
39:  cycle?: DependencyPeriodType
40:  dateValue?: string
41:  dependResult?: 'success' | 'failed'
42:  parameterPassing?: boolean
43:}
44:
45:/** 组内条件按 join 连接；多组之间约定为 AND（草案语义，后端冻结前不改） */
46:export interface DependencyGroup {
47:  join: 'AND' | 'OR'
48:  conditions: DependencyCondition[]
49:}
50:
51:/* ---------- 表达式预览 / 判定 ---------- */
52:
53:export type DependencyVerdict = 'satisfied' | 'pending' | 'failed' | 'timeout'
54:
55:export interface DependencyExpressionPreview {
56:  /** 服务端拼装的最终依赖表达式（执行前所见即执行所判） */
57:  expression: string
58:  resolved: Array<{
59:    upstreamWorkflowId: number
60:    upstreamTaskCode?: string
61:    windowStart: string
62:    windowEnd: string
63:    verdict: DependencyVerdict
64:  }>
65:  verdict: DependencyVerdict
66:}
67:
68:export interface DependencySaveInput {
69:  projectId: number
70:  /** 依赖配置随工作流版本保存（客户 #2/#5 同语义） */
71:  workflowId: number
72:  workflowVersion: number
73:  taskId: string
74:  groups: DependencyGroup[]
75:  groupRelation?: 'AND' | 'OR'
76:  checkIntervalSeconds?: number
77:  failurePolicy?: 'fail' | 'waiting'
78:  failureWaitingSeconds?: number
79:  timeoutSeconds?: number
80:  dependentWaitTimeoutEnabled?: boolean
81:  dependentWaitTimeoutSeconds?: number
82:  dependentWaitTimeoutStrategies?: Array<'warn' | 'failed'>
83:}
84:
85:export interface DependencyEvaluateRequest {
86:  projectId: number
87:  workflowId: number
88:  workflowVersion: number
89:  taskId: string
90:  groups: DependencyGroup[]

92:  asOf?: string
93:  groupRelation?: 'AND' | 'OR'
94:  timeoutSeconds?: number
95:}
